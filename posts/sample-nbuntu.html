Java

public class CoffeeMachine {
    private int waterLevel;
    private boolean isPowered;

    public CoffeeMachine(int waterLevel) {
        this.waterLevel = waterLevel;
        this.isPowered = false;
    }

    public String brew(String type) {
        if (!isPowered) {
            return "전원부터 켜세요...";
        }
        if (waterLevel < 200) {
            throw new RuntimeException("물 부족! 물부터 채우세요.");
        }
        waterLevel -= 200;
        return type + " 한 잔 완성! (남은 물: " + waterLevel + "ml)";
    }
}

Kotlin

data class Developer(
    val name: String,
    val languages: List<String>,
    val caffeineLevel: Int = 100
) {
    fun introduce(): String {
        val status = when {
            caffeineLevel > 80 -> "코딩 머신 모드"
            caffeineLevel > 40 -> "그럭저럭 작동중"
            else -> "커피 필요... 긴급..."
        }
        return "$name ($status) - ${languages.joinToString()}"
    }

    fun drinkCoffee() = copy(
        caffeineLevel = (caffeineLevel + 30).coerceAtMost(100)
    )
}

Python

from dataclasses import dataclass
from typing import Optional

@dataclass
class TodoItem:
    title: str
    done: bool = False
    priority: int = 0

    def toggle(self) -> "TodoItem":
        return TodoItem(self.title, not self.done, self.priority)

def find_urgent(todos: list[TodoItem]) -> Optional[TodoItem]:
    """긴급한 미완료 항목을 찾습니다. 없으면 퇴근."""
    pending = [t for t in todos if not t.done and t.priority > 5]
    return max(pending, key=lambda t: t.priority) if pending else None

의사코드 (CLRS 스타일)

Introduction to Algorithms 스타일의 의사코드를 한글로도 작성할 수 있습니다.

영문 의사코드

BINARY-SEARCH(A, v, low, high)
  if low > high
    return NIL
  mid = (low + high) / 2
  if A[mid] == v
    return mid
  else if A[mid] > v
    return BINARY-SEARCH(A, v, low, mid - 1)
  else
    return BINARY-SEARCH(A, v, mid + 1, high)

한글 의사코드

이진탐색(A, v, low, high)
  만약 low > high
    반환 NIL
  mid ← (low + high) / 2
  만약 A[mid] == v
    반환 mid
  아니면 만약 A[mid] > v
    반환 이진탐색(A, v, low, mid - 1)
  아니면
    반환 이진탐색(A, v, mid + 1, high)

// 시간 복잡도: O(log n)
// 아침에 커피 찾는 것보다 빠릅니다
