Java 코드를 포함한 코드 블록입니다.

@Service
public class CoffeeService {
    private final CoffeeRepository repository;

    public CoffeeService(CoffeeRepository repository) {
        this.repository = repository;
    }

    public List<Coffee> findAll() {
        return repository.findAll();
    }

    public Coffee brew(String type) {
        if (type == null || type.isBlank()) {
            throw new IllegalArgumentException("커피 타입을 지정해주세요");
        }
        return new Coffee(type, LocalDateTime.now());
    }
}

from dataclasses import dataclass
from typing import Optional

@dataclass
class Developer:
    name: str
    caffeine_level: int = 100

    def need_coffee(self) -> bool:
        return self.caffeine_level < 30

    def drink(self) -> "Developer":
        return Developer(self.name, min(100, self.caffeine_level + 40))
